  <script>
  (function(){
    const root = document.querySelector('#aquablast-{{ sec_id }}');
    if(!root) return;

    // Sync quantity inputs across forms
    const qty = root.querySelector('#qty-{{ sec_id }}');
    const qtyTargets = root.querySelectorAll('[data-qty-input]');
    function syncQty(){ qtyTargets.forEach(i => i.value = Math.max(1, parseInt(qty?.value||'1',10))); }
    qty?.addEventListener('input', syncQty); syncQty();

    // Variant logic (vanilla): match selected options to variant ID
    const productJson = {{ section.settings.product | json }};
    const selectors = root.querySelectorAll('select[name^="options["]');
    const variantInputs = root.querySelectorAll('[data-variant-input]');
    const heroImg = root.querySelector('#heroImg-{{ sec_id }}');
    const skuView = root.querySelector('#skuView-{{ sec_id }}');

    function currentOptions(){
      const opts = [];
      selectors.forEach(s => { const idx = parseInt(s.dataset.optionIndex,10); opts[idx] = s.value; });
      return opts;
    }
    function findVariant(opts){
      if(!productJson || !productJson.variants) return null;
      return productJson.variants.find(v => {
        return v.options.every((val, i) => String(val) === String(opts[i]));
      }) || productJson.variants[0] || null;
    }
    function updateVariant(){
      const v = findVariant(currentOptions());
      if(!v) return;
      variantInputs.forEach(i => i.value = v.id);
      if(skuView) skuView.textContent = 'SKU: ' + (v.sku || v.id);
      if(heroImg && v.featured_image && v.featured_image.src){ heroImg.src = v.featured_image.src; }
    }
    selectors.forEach(sel => sel.addEventListener('change', updateVariant));
    updateVariant();

    // Countdown: N days from first visit (per-section key)
    const DAYS = {{ section.settings.promo_days | default: 7 | plus: 0 }};
    const KEY = 'aquablast_promo_start_ts_{{ sec_id }}';
    const WEEK = DAYS*24*60*60*1000;
    let start = Number(localStorage.getItem(KEY));
    if(!start){ start = Date.now(); localStorage.setItem(KEY, String(start)); }
    const deadline = start + WEEK;
    const main = root.querySelector('#dealTimer-{{ sec_id }}');
    const sticky = root.querySelector('#dealTimerSticky-{{ sec_id }}');
    function pad(n){ return String(n).padStart(2,'0'); }
    function tick(){
      const now = Date.now();
      let diff = deadline - now;
      if(diff <= 0){
        main && (main.textContent = 'המבצע הסתיים');
        sticky && (sticky.textContent = 'המבצע הסתיים');
        root.querySelectorAll('.cta').forEach(btn => btn.disabled = true);
        clearInterval(t);
        return;
      }
      const d = Math.floor(diff / (24*60*60*1000)); diff %= 24*60*60*1000;
      const h = Math.floor(diff / (60*60*1000)); diff %= 60*60*1000;
      const m = Math.floor(diff / (60*1000)); diff %= 60*1000;
      const s = Math.floor(diff / 1000);
      const txt = `${d} ימים ${pad(h)}:${pad(m)}:${pad(s)}`;
      main && (main.textContent = txt);
      sticky && (sticky.textContent = txt);
    }
    tick(); const t = setInterval(tick, 1000);
  })();
  </script>
